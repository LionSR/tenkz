% Regression: an atom threaded on a wire stands at no cell, so no endpoint
% names it and every face looks free.  Its carrier runs through the bead,
% and the faces that run crosses are the ones the automatic station has to
% avoid.  How many it crosses depends on where along the carrier the bead
% stands: at either end of the route the carrier only leaves the bead and
% one face carries ink, so reserving both would give up a free face.
\documentclass{standalone}

\int_new:N \l__tenkz_test_side_count_int

\dim_new:N \l__tenkz_test_bead_dx_dim
\dim_new:N \l__tenkz_test_bead_dy_dim
\cs_new_eq:NN
  \__tenkz_test_side_auto:nN \__tenkz_kernel_r_label_auto_side:nN
\cs_new_protected:Npn \__tenkz_test_stationary_bead:
  {
    \prop_put:Nnn \l__tenkz_string_kind_prop { stationary-probe } { open }
    \path [ spath/save = stationary-probe ]
      (0,0) .. controls (1,0) and (0,0) .. (1,0) ;
    \__tenkz_string_bead_frame:nnnNN
      {stationary-probe} {0.5} {tenkz-stationary-probe}
      \l__tenkz_test_bead_dx_dim \l__tenkz_test_bead_dy_dim
    \dim_compare:nNnF
      {
        \dim_abs:n { \l__tenkz_test_bead_dx_dim }
        + \dim_abs:n { \l__tenkz_test_bead_dy_dim }
      }
      = {0pt}
      { \errmessage{stationary~cubic~bead~returned~a~direction} }
  }
\cs_new_eq:NN \tenkzteststationarybead \__tenkz_test_stationary_bead:
\cs_set_protected:Npn \__tenkz_kernel_r_label_auto_side:nN #1#2
  {
    \__tenkz_test_side_auto:nN {#1} #2
    \int_gincr:N \l__tenkz_test_side_count_int
    \__tenkz_model_get:nnN {#1} {label} \l__tenkz_test_side_label_tl
    \str_case:VnF \l__tenkz_test_side_label_tl
      {
        {D}
          {
            \str_if_eq:VnF #2 {e}
              { \errmessage{a~bead~on~a~column~wire~kept~its~south} }
          }
        {E}
          {
            \str_if_eq:VnF #2 {s}
              { \errmessage{a~bead~on~a~row~wire~lost~its~free~south} }
          }
        {F}
          {
            \str_if_eq:VnF #2 {n}
              { \errmessage{a~bead~at~the~head~of~a~route~gave~up~north} }
          }
        {G}
          {
            \str_if_eq:VnF #2 {s}
              { \errmessage{a~bead~at~the~tail~of~a~route~gave~up~south} }
          }
        {K}
          {
            \str_if_eq:VnF #2 {n}
              { \errmessage{a~routed~head~bead~lost~its~forward~tangent} }
          }
        {L}
          {
            \str_if_eq:VnF #2 {s}
              { \errmessage{a~routed~tail~bead~lost~its~reverse~tangent} }
          }
        {H}
          {
            \str_if_eq:VnF #2 {e}
              { \errmessage{a~leg~carrier~left~the~face~it~grows~from~free} }
          }
        {J}
          {
            \str_if_eq:VnF #2 {e}
              { \errmessage{a~one-ended~carrier~reserved~no~face~at~all} }
          }
      }
      { \errmessage{unexpected~label~in~on-wire~fixture} }

% A carrier down the column: north and south carry it, east is free.
\begin{tenkz}[lattice={3x1}, bonds=none]
  \tn[at=(1,1), name=P]{}
  \tn[at=(3,1), name=Q]{}
  \tnwire[name=W]{P}{Q}
  \tn[at=on W 0.5, skin=dot]{D}
\end{tenkz}
% A carrier along the row: east and west carry it, south stays free.
\begin{tenkz}[lattice={1x3}, bonds=none]
  \tn[at=(1,1), name=R]{}
  \tn[at=(1,3), name=S]{}
  \tnwire[name=Y]{R}{S}
  \tn[at=on Y 0.5, skin=dot]{E}
\end{tenkz}
% At the head of the route the carrier only runs south out of the bead, so
% north stays free and the first free face in s, n, e, w is north.
\begin{tenkz}[lattice={3x1}, bonds=none]
  \tn[at=(1,1), name=P0]{}
  \tn[at=(3,1), name=Q0]{}
  \tnwire[name=W0]{P0}{Q0}
  \tn[at=on W0 0, skin=dot]{F}
\end{tenkz}
% At the tail the carrier only arrives from the north, so south stays free.
\begin{tenkz}[lattice={3x1}, bonds=none]
  \tn[at=(1,1), name=P1]{}
  \tn[at=(3,1), name=Q1]{}
  \tnwire[name=W1]{P1}{Q1}
  \tn[at=on W1 1, skin=dot]{G}
\end{tenkz}
% Smooth routed strings use the same endpoint incidence count.  This pins the
% forward sign at the head and its reversal at the tail; the curved midpoint
% fixture in r_label_auto_gaps.tex separately distinguishes tangent from chord.
\begin{tenkz}[lattice={3x1}, bonds=none]
  \tnwire[kind=string, name=BE, via={(2,1)}]{open n}{open s}
  \tn[at=on BE 0, skin=dot]{K}
  \tn[at=on BE 1, skin=dot]{L}
\end{tenkz}
% A policy leg carries its bead too: the leg grows south out of its host, so
% the bead's north and south both carry ink and east is the first face free.
\begin{tenkz}[rows={wire}, cols=2, physical=down]
  \tn{} & \tn{}
  \tn[at=on leg-s-1-1 0.5, skin=dot]{H}
\end{tenkz}
% A carrier with one authored end and one open side runs off the picture by
% the word that side names, so the bead's north and south both carry it.
\begin{tenkz}[lattice={1x1}, bonds=none]
  \tn[at=(1,1), name=Z]{}
  \tnwire[name=WZ]{Z}{open s}
  \tn[at=on WZ 0.5, skin=dot]{J}
\end{tenkz}
