% Regression: automatic stations read route incidence from resolved geometry.
%
% Endpoint bearings and rendered waypoint tangents are pulled back through the
% owning atom's complete carrier basis.  The same reading covers non-cell
% placements, basis members, closed routes, generated hull departures, and
% circle frames.
% Circle bearings still choose a fallback position, but the frame-wide guard
% withholds their claims because a nonincident chord can cross the band.  The
% The remaining blind pins distinguish unsupported span-internal geometry
% from the later label-order step of issue #15.

% A named endpoint whose atom stands a step off the frame owns no cell.  Its
% resolved bearing to C nevertheless occupies east, so its historical south
% station is now a verified automatic claim.
\begin{tenkz}[lattice={2x2}, bonds=none]
  \tn[at=(1,1), name=A]{}
  \tn[at=1 s of A, name=P, skin=dot]{P}
  \tn[at=(2,2), name=C]{}
  \tnwire{P}{C}
\end{tenkz}
% The same off-cell reading is carrier-local.  This plane sends logical east
% toward page southwest; G still reserves its local east face and keeps south.
\begin{tenkz}[lattice={2x2}, frame=plane, bonds=none]
  \tn[at=(1,1), name=GA]{}
  \tn[at=1 s of GA, name=G, skin=dot]{G}
  \tn[at=(2,2), name=GC]{}
  \tnwire{G}{GC}
\end{tenkz}
% A waypoint spelt as a relative address names a point, not a cell.  The
% resolved first segment leaves Q south, rather than following the eastward
% endpoint chord, so Q moves north.
\begin{tenkz}[lattice={2x3}, bonds=none]
  \tn[at=(1,1), name=Q, skin=dot]{Q}
  \tn[at=(1,3), name=E]{}
  \tnwire[kind=string, via={1 s of Q}]{Q}{E}
\end{tenkz}
% A non-cell endpoint on a smooth string must read the rendered endpoint
% tangent, not the chord to its first authored waypoint.  The southeast chord
% would occupy east and leave L south; the Hobby tangent occupies south, so L
% moves north and can make a verified claim.
\begin{tenkz}[lattice={3x4}, bonds=none]
  \tn[at=(1,1), name=LA]{}
  \tn[at=1 s of LA, name=L, skin=dot]{L}
  \tn[at=2 e of L, name=LE]{}
  \tnwire[kind=string, via={1 se of L}]{L}{LE}
\end{tenkz}
% An authored waypoint at either end of the via list is still interior when
% the string opens on that side.  A is the sole via between two open ends, so
% both frozen tangents must be read; north and south are occupied and A moves
% east.  Skipping either boundary leaves an earlier fallback available.
\begin{tenkz}[lattice={1x1}, bonds=none]
  \tn[at=(1,1), name=OPEN, skin=dot]{A}
  \tnwire[kind=string, via={OPEN}]{open n}{open s}
\end{tenkz}
% Open words are renderer-owned directions, including diagonals.  D stands
% off-cell and has a south stub plus a northeast stub: south and north are
% occupied, so its first free verified station is east.  Reading only the
% four cardinal words would leave the northeast ink falsely free.
\begin{tenkz}[lattice={1x1}, bonds=none]
  \tn[at=(1,1), name=DA]{}
  \tn[at=1 s of DA, name=D, skin=dot]{D}
  \tnwire{D}{open s}
  \tnwire{D}{open ne}
\end{tenkz}
% The endpoint names I's west cell, while its first geometric waypoint lies
% at the east cell inside the same three-cell span.  Measuring containment
% from the incidence cell would miss that hidden two-cell run; the atom centre
% keeps I on its deterministic south fallback and in the blind ledger.
\begin{tenkz}[lattice={4x5}, bonds=none]
  \tn[at=(3,2), wide=3, name=I, skin=dot]{I}
  \tn[at=(1,4), name=IE]{}
  \tnwire[kind=string, route=orth, via={1 e of I}]{(3,2)}{IE}
\end{tenkz}
% The same span-internal neighbour can follow a labelled interior waypoint.
% The path enters O visibly from the west, then turns north at the internal
% east point; that hidden side is disabled and makes the station unverified.
\begin{tenkz}[lattice={4x5}, bonds=none]
  \tn[at=(3,2), wide=3, name=O, skin=dot]{O}
  \tn[at=(3,1), name=OF]{}
  \tn[at=(1,4), name=OE]{}
  \tnwire[kind=string, route=orth, via={O,1 e of O}]{OF}{OE}
\end{tenkz}
% An atom named as an interior orthogonal waypoint is entered from north and
% left south, even though neither endpoint names it; east is its first free
% station.  The frozen path, rather than the endpoint chord, supplies both
% rendered line tangents.
\begin{tenkz}[lattice={3x1}, bonds=none]
  \tn[at=(1,1), name=F]{}
  \tn[at=(2,1), name=R, skin=dot]{R}
  \tn[at=(3,1), name=H]{}
  \tnwire[kind=string, route=orth, via={(2,1)}]{F}{H}
\end{tenkz}
% A bead on a string that bends through its own waypoints reads the rendered
% tangent at its own arclength.  This half-turn-symmetric Hobby route gives the
% midpoint a north/south-bearing tangent, so those faces are occupied and T
% moves east.  Its endpoint chord is horizontal and would leave T south.
\begin{tenkz}[lattice={3x3}, bonds=none]
  \tn[at=(2,1), name=J]{}
  \tn[at=(2,3), name=K]{}
  \tnwire[kind=string, name=RS, via={(1,2),(2,2),(3,2)}]{J}{K}
  \tn[at=on RS 0.5, skin=dot]{T}
\end{tenkz}
% Two basis members of one outer cell share its row and column, but their
% resolved member points differ in the plane basis.  M occupies local north
% and stays south; N occupies local south and moves north.
\begin{tenkz}[lattice={1x2}, bonds=none,
    frame={plane, basis={wire at (0,0), wire at (0,2)}}]
  \tn[at=(1,1,1), skin=dot]{M}
  \tn[at=(1,1,2), skin=dot]{N}
  \tnwire{(1,1,1)}{(1,1,2)}
\end{tenkz}
% A circle closes its columns into a ring, so column arithmetic cannot read a
% chord.  In each endpoint's own tangent/outward basis this chord leaves
% inward through local south, and both labels would move north.  U's explicit
% same-dot mark takes north first, so U moves local east while V stays north.
% Their positions are resolved, but the frame-wide guard withholds verified
% claims because a different, nonincident chord can cross either label band.
\begin{tenkz}[rows={wire}, cols=4, bonds=none, frame=circle]
  \tn[at=(1,1), name=CU, skin=dot]{U}
  \tn[at=(1,3), skin=dot]{V}
  \tnwire{(1,1)}{(1,3)}
  \tnmark[form=label, label pos=n]{CU}{$c$}
\end{tenkz}
% A bead on a circle carrier reads the already-rendered local tangent.  The
% vertical chord occupies north and south, so W takes east.  Circle-wide claim
% certification remains suppressed independently by the frame guard.
\begin{tenkz}[rows={wire}, cols=4, bonds=none, frame=circle]
  \tn[at=(1,1), name=C1]{}
  \tn[at=(1,3), name=C3]{}
  \tnwire[name=CW]{C1}{C3}
  \tn[at=on CW 0.5, skin=dot]{W}
\end{tenkz}
% The same local reading applies to a smooth circle string with an authored
% waypoint.  The rightmost midpoint has a north/south tangent, so B also takes
% east without turning the frame-wide answer into a verified claim.
\begin{tenkz}[rows={wire}, cols=4, bonds=none, frame=circle]
  \tn[at=(1,1), name=V1]{}
  \tn[at=(1,3), name=V3]{}
  \tnwire[kind=string, name=CV, via={(1,2)}]{V1}{V3}
  \tn[at=on CV 0.5, skin=dot]{B}
\end{tenkz}
% An explicit empty waypoint list is the same route as an omitted list.  Even
% with route=orth there is no corner here, so the circle tangent remains
% answerable and C takes east.
\begin{tenkz}[rows={wire}, cols=4, bonds=none, frame=circle]
  \tn[at=(1,1), name=E1]{}
  \tn[at=(1,3), name=E3]{}
  \tnwire[kind=string, name=CE, route=orth, via={}]{E1}{E3}
  \tn[at=on CE 0.5, skin=dot]{C}
\end{tenkz}
% A fixed cardinal mark station on the exact same named ordinary dot is known
% before either label is measured.  The wire takes south and the explicit
% mark takes north, so X's automatic station moves east.  Indirect hosts,
% arbitrary bearings, and auto-versus-auto ordering still need the general
% measured-box step of issue #15.
\begin{tenkz}[lattice={2x1}, bonds=none]
  \tn[at=(1,1), name=XA, skin=dot]{X}
  \tn[at=(2,1), name=XB]{}
  \tnwire{XA}{XB}
  \tnmark[form=label, label pos=90]{XA}{$m$}
\end{tenkz}
% A fixed diagonal is not a face reservation.  South is occupied, but the
% mark at 45 degrees leaves H's north station available; sector-mapping the
% mark would incorrectly move H east.
\begin{tenkz}[lattice={2x1}, bonds=none]
  \tn[at=(1,1), name=HA, skin=dot]{H}
  \tn[at=(2,1), name=HB]{}
  \tnwire{HA}{HB}
  \tnmark[form=label, label pos=45]{HA}{$h$}
\end{tenkz}
% A closed Hobby string has no endpoints.  Its rendered seam tangents differ
% from the two waypoint chords, and the labelled northwest waypoint takes
% east.  The zero-length soft-path close component must not make it blind.
\begin{tenkz}[lattice={3x3}, bonds=none]
  \tn[at=(1,1), skin=dot]{Y}
  \tnwire[kind=string, closed,
    via={(1,1),(3,1),(3,3),(1,3)}]
\end{tenkz}
% A hull route's first generated point is not present in from/via/to.  From Z
% the frozen route leaves through local west toward the selected hull, while
% the ordinary chord to ZE runs east.  Z stands above the selected hull, whose
% endpoint tangent lies in that sector.  Two vertical probes take north
% and south, leaving east as the first free station; the endpoint chord alone
% would instead take east and force the name west.
\begin{tenkz}[lattice={5x4}, bonds=none]
  \tn[at=(3,2), skin=box]{}
  \tn[at=(3,3), skin=box]{}
  \tn[at=(4,2), skin=box]{}
  \tn[at=(4,3), skin=box]{}
  \tn[at=(1,2), name=Z, skin=dot]{Z}
  \tn[at=(1,4), name=ZE]{}
  \tn[at=1 n of Z, name=ZN]{}
  \tn[at=(2,2), name=ZS]{}
  \tnwire[name=zn]{Z}{ZN}
  \tnwire[name=zs]{Z}{ZS}
  \tnwire[kind=string, name=hz, route={n of (3,2) .. (4,3)},
    crossing=over,
    cross={over at crossing of hz and zn, over at crossing of hz and zs}]
    {Z}{ZE}
\end{tenkz}
